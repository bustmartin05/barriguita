📋 Código SQL para tu Base de Datos

Si aún no creaste las tablas en Supabase, copia este script y pégalo en el SQL Editor de Supabase: 

-- TABLAS Y POLÍTICAS DE ACCESO PARA BARRIGUITAS
CREATE TABLE IF NOT EXISTS barriguitas_prices (
  id TEXT PRIMARY KEY,
  category TEXT NOT NULL,
  item_key TEXT NOT NULL,
  price NUMERIC NOT NULL,
  metadata TEXT,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

CREATE TABLE IF NOT EXISTS barriguitas_shipping_zones (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  price NUMERIC NOT NULL,
  active BOOLEAN DEFAULT true,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

CREATE TABLE IF NOT EXISTS barriguitas_upsell_offers (
  id TEXT PRIMARY KEY,
  trigger_category TEXT NOT NULL DEFAULT 'all',
  message TEXT NOT NULL,
  product_name TEXT NOT NULL,
  discount_price NUMERIC NOT NULL,
  original_price NUMERIC DEFAULT 0,
  active BOOLEAN DEFAULT true,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

CREATE TABLE IF NOT EXISTS barriguitas_promos (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  badge TEXT,
  title TEXT NOT NULL,
  description TEXT NOT NULL,
  price NUMERIC NOT NULL,
  active BOOLEAN DEFAULT true,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

CREATE TABLE IF NOT EXISTS barriguitas_gallery (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  title TEXT NOT NULL,
  category TEXT NOT NULL,
  image_url TEXT NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

CREATE TABLE IF NOT EXISTS barriguitas_reviews (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  name TEXT NOT NULL,
  rating INTEGER DEFAULT 5,
  date_text TEXT,
  text TEXT NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

ALTER TABLE barriguitas_prices ENABLE ROW LEVEL SECURITY;
ALTER TABLE barriguitas_shipping_zones ENABLE ROW LEVEL SECURITY;
ALTER TABLE barriguitas_upsell_offers ENABLE ROW LEVEL SECURITY;
ALTER TABLE barriguitas_promos ENABLE ROW LEVEL SECURITY;
ALTER TABLE barriguitas_gallery ENABLE ROW LEVEL SECURITY;
ALTER TABLE barriguitas_reviews ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Public access for prices" ON barriguitas_prices;
CREATE POLICY "Public access for prices" ON barriguitas_prices FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Public access for shipping" ON barriguitas_shipping_zones;
CREATE POLICY "Public access for shipping" ON barriguitas_shipping_zones FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Public access for upsell" ON barriguitas_upsell_offers;
CREATE POLICY "Public access for upsell" ON barriguitas_upsell_offers FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Public access for promos" ON barriguitas_promos;
CREATE POLICY "Public access for promos" ON barriguitas_promos FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Public access for gallery" ON barriguitas_gallery;
CREATE POLICY "Public access for gallery" ON barriguitas_gallery FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Public access for reviews" ON barriguitas_reviews;
CREATE POLICY "Public access for reviews" ON barriguitas_reviews FOR ALL USING (true) WITH CHECK (true);
